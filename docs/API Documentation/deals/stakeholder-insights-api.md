# Deal Stakeholder Insights API

This API manages AI-generated stakeholder insights for CRM deals. Insights are automatically generated by analyzing meeting transcripts, stakeholder engagement patterns, and deal context.

**Base URL**: `/users/crm/deals`

---

## Overview

### What are Stakeholder Insights?

AI-generated recommendations about stakeholder engagement, risks, and opportunities. These insights help sales teams take proactive action on deals.

| Insight Type | Description | Example |
| --- | --- | --- |
| **ENGAGEMENT_GAP** | Key stakeholder not contacted recently | "Decision maker hasn't been engaged in 3 weeks" |
| **SENTIMENT_DECLINING** | Champion/supporter enthusiasm dropping | "Sarah's engagement score dropped from 85 to 62" |
| **CHAMPION_OPPORTUNITY** | High-potential stakeholder identified | "John shows strong champion potential (score: 85)" |
| **STAKEHOLDER_RISK** | Red flags detected | "Champion mentioned evaluating competitor" |
| **MISSING_DECISION_MAKER** | No decision maker in stakeholder list | "No stakeholder with decision-making authority" |
| **COVERAGE_GAP** | Key department not represented | "IT department not represented in stakeholders" |
| **UNADDRESSED_CONCERN** | Open concern not followed up | "Security concern raised in meeting not addressed" |

### Insight Lifecycle

```
AI Generation Flow:
(analysis runs) → ACTIVE → (user sees) → ACKNOWLEDGED → (action taken) → RESOLVED
                                                     ↘ (not relevant) → DISMISSED
                  ↓
            (time passes) → EXPIRED
```

### AI vs Manual Items

| Origin | Initial Status | Generation Trigger |
| --- | --- | --- |
| **AI-generated** | `ACTIVE` | Stakeholder analysis runs (background) |

> **Note**: Insights are generated automatically during stakeholder analysis. There is no manual creation endpoint - use Risks & Blockers API for manual items.

---

## Response Envelope

All endpoints return the shared `GenericResponse` envelope:

```json
{
  "status": "SUCCESS | ERROR",
  "message": "Human-readable message",
  "data": { ... } | null
}
```

---

## Authentication

- Bearer token required (`Authorization: Bearer <access_token>`)
- Organisation/user/role are inferred from the token

---

## Rate Limiting

| Operation | Limit |
| --- | --- |
| Read (`GET`) | ~60/min, 1000/hour per user |
| Write (`POST`) | ~30/min, 500/hour per user |

On limit breach: `429 Too Many Requests` with `Retry-After` header.

## Data Shapes (TypeScript)

```ts
export type UUID = string;

// ============================================================================
// Enums
// ============================================================================

type StakeholderInsightType =
  | "ENGAGEMENT_GAP"        // Key person not contacted recently
  | "SENTIMENT_DECLINING"   // Champion/supporter sentiment dropping
  | "CHAMPION_OPPORTUNITY"  // Potential new champion identified
  | "STAKEHOLDER_RISK"      // Red flags (competitor mention, stance shift)
  | "MISSING_DECISION_MAKER"// No decision maker in stakeholder list
  | "COVERAGE_GAP"          // Key department not represented
  | "UNADDRESSED_CONCERN";  // Open concern not followed up

type StakeholderInsightStatus =
  | "ACTIVE"                // Currently relevant, needs attention
  | "ACKNOWLEDGED"          // User has seen it
  | "RESOLVED"              // Action taken, issue addressed
  | "DISMISSED"             // User dismissed as not relevant
  | "EXPIRED";              // No longer relevant (stale)

type StakeholderInsightPriority =
  | "CRITICAL"              // Red - immediate action required
  | "HIGH"                  // Orange - attention required this week
  | "MEDIUM"                // Yellow - worth reviewing soon
  | "LOW";                  // Blue - informational

type InsightActionType =
  | "SCHEDULE_CALL"         // Schedule a call with stakeholder
  | "SEND_EMAIL"            // Draft/send an email
  | "VIEW_DETAILS"          // View detailed information
  | "VIEW_PROFILE"          // View stakeholder profile
  | "GENERATE_BRIEFING";    // Generate a briefing document

// ============================================================================
// Response Models
// ============================================================================

/** Action button for the insight card */
export interface InsightAction {
  action_type: InsightActionType;
  label: string;                    // Display text (e.g., "Schedule Call")
  parameters: Record<string, any>;  // Additional params (stakeholder_id, etc.)
}

/** Single insight response */
export interface StakeholderInsightResponse {
  insight_id: UUID;
  insight_type: StakeholderInsightType;
  priority: StakeholderInsightPriority;
  status: StakeholderInsightStatus;

  // Content
  title: string;                    // Short actionable title
  description: string | null;       // Detailed explanation
  ai_reasoning: string;             // Why this insight was generated

  // Stakeholder context (null for deal-level insights)
  stakeholder_id: UUID | null;
  stakeholder_name: string | null;
  stakeholder_title: string | null;
  stakeholder_email: string | null;

  // Actions
  actions: InsightAction[];

  // Metrics (evidence)
  metric_value: number | null;      // e.g., 21 (days)
  metric_unit: string | null;       // e.g., "days"
  threshold_value: number | null;   // e.g., 14 (threshold exceeded)
  trend_direction: string | null;   // "declining", "improving", "stable"

  // AI metadata
  ai_confidence: number;            // 0.0 - 1.0

  // Source
  source_type: string | null;       // "agent", "analysis_run"
  related_risk_blocker_id: UUID | null;

  // Timestamps
  identified_at: string;            // ISO8601 - when insight was detected
  expires_at: string | null;        // ISO8601 - when insight becomes stale
  created_at: string;               // ISO8601
  updated_at: string;               // ISO8601
}

/** Summary statistics */
export interface StakeholderInsightSummary {
  critical_count: number;
  high_count: number;
  medium_count: number;
  low_count: number;
}

/** List response */
export interface StakeholderInsightListResponse {
  insights: StakeholderInsightResponse[];
  total_count: number;
  summary: StakeholderInsightSummary;
}
```

## 1) List Stakeholder Insights

Retrieve AI-generated insights for a deal's stakeholders.

- **Endpoint**: `GET /{deal_id}/stakeholder-insights`
- **Tags**: `stakeholder-insights`

### Query Parameters

| Name | Type | Default | Description |
| ---- | ---- | ------- | ----------- |
| `limit` | int (1-50) | 20 | Number of insights to return |
| `status` | string | null | Comma-separated statuses: `ACTIVE,ACKNOWLEDGED` |
| `priority` | enum | null | Filter: `CRITICAL`, `HIGH`, `MEDIUM`, `LOW` |
| `insight_type` | enum | null | Filter by insight type |
| `include_expired` | boolean | false | Include expired/resolved/dismissed insights |

### Sample Request

```http
GET /users/crm/deals/6c3f98d5-2e6b-4a1c-8f8c-2e8f0211b0b9/stakeholder-insights?limit=20&status=ACTIVE
Authorization: Bearer <token>
```

## 5) Resolve Insight

Mark an insight as resolved (action has been taken).

- **Endpoint**: `POST /{deal_id}/stakeholder-insights/{insight_id}/resolve`
- **Tags**: `stakeholder-insights`

### Path Parameters

| Name | Type | Description |
| ---- | ---- | ----------- |
| `deal_id` | UUID | Deal identifier |
| `insight_id` | UUID | Insight identifier |

### Request Body (Optional)

```json
{
  "resolution_notes": "Scheduled call with Arun for next Tuesday. He confirmed continued interest."
}
```

| Field | Type | Required | Description |
| ----- | ---- | -------- | ----------- |
| `resolution_notes` | string (max 2000) | No | Notes about how the insight was addressed |

### Sample Response

```json
{
  "status": "SUCCESS",
  "message": "Insight resolved successfully",
  "data": {
    "insight_id": "f9a6c1e1-0d56-4e2b-9c2d-4d3fb3d6c9a1",
    "status": "RESOLVED",
    ...
  }

```

### cURL Example

| HTTP | When |
| ---- | ---- |
| `400` | Invalid request body, invalid status transition |
| `401` | Missing/invalid bearer token |
| `403` | Not authorized for this org/deal |
| `404` | Deal or insight not found |
| `429` | Rate limit exceeded |
| `500` | Unexpected server error |

## Frontend Integration Example

```typescript
// Types (from above)
interface StakeholderInsightResponse { ... }
interface StakeholderInsightListResponse { ... }

// ============================================================================
// API Client
// ============================================================================

class StakeholderInsightsAPI {
  constructor(private baseUrl: string, private token: string) {}

  private async request<T>(path: string, options?: RequestInit): Promise<T> {
    const res = await fetch(`${this.baseUrl}${path}`, {
      ...options,
      headers: {
        'Authorization': `Bearer ${this.token}`,
        'Content-Type': 'application/json',
        ...options?.headers,
      },
    });
    const json = await res.json();
    if (!res.ok) throw new Error(json.message);
    return json.data;
  }

  // List insights with filters
  async list(dealId: string, params?: {
    limit?: number;
    status?: string;
    priority?: string;
    insight_type?: string;
    include_expired?: boolean;
  }): Promise<StakeholderInsightListResponse> {
    const query = new URLSearchParams(
      Object.entries(params || {})
        .filter(([_, v]) => v !== undefined)
        .map(([k, v]) => [k, String(v)])
    );
    return this.request(`/deals/${dealId}/stakeholder-insights?${query}`);
  }

  // Get single insight
  async get(insightId: string): Promise<StakeholderInsightResponse> {
    return this.request(`/stakeholder-insights/${insightId}`);
  }

  // Status transitions
  async acknowledge(dealId: string, insightId: string): Promise<StakeholderInsightResponse> {
    return this.request(`/deals/${dealId}/stakeholder-insights/${insightId}/acknowledge`, {
      method: 'POST',
    });
  }

  async dismiss(dealId: string, insightId: string): Promise<StakeholderInsightResponse> {
    return this.request(`/deals/${dealId}/stakeholder-insights/${insightId}/dismiss`, {
      method: 'POST',
    });
  }

  async resolve(
    dealId: string,
    insightId: string,
    notes?: string
  ): Promise<StakeholderInsightResponse> {
    return this.request(`/deals/${dealId}/stakeholder-insights/${insightId}/resolve`, {
      method: 'POST',
      body: JSON.stringify({ resolution_notes: notes }),
    });
  }
}

// ============================================================================
// Usage Example - AI Insights Panel
// ============================================================================

const api = new StakeholderInsightsAPI('/api/v1/users/crm', accessToken);

// Fetch active insights for the deal
const { insights, summary } = await api.list(dealId, {
  status: 'ACTIVE,ACKNOWLEDGED',
  limit: 20,
});

// Render summary badges
console.log(`Critical: ${summary.critical_count}`);
console.log(`High: ${summary.high_count}`);

// Render insight cards grouped by priority
const critical = insights.filter(i => i.priority === 'CRITICAL');
const high = insights.filter(i => i.priority === 'HIGH');
const medium = insights.filter(i => i.priority === 'MEDIUM');

// Render a single insight card
function renderInsightCard(insight: StakeholderInsightResponse) {
  return {
    title: insight.title,
    stakeholder: insight.stakeholder_name
      ? `${insight.stakeholder_name} (${insight.stakeholder_title})`
      : 'Deal-level insight',
    description: insight.description,
    reasoning: insight.ai_reasoning,
    actions: insight.actions.map(a => ({
      label: a.label,
      onClick: () => handleAction(a.action_type, a.parameters),
    })),
    priorityColor: {
      CRITICAL: 'red',
      HIGH: 'orange',
      MEDIUM: 'yellow',
      LOW: 'blue',
    }[insight.priority],
  };
}

// Handle action button clicks
function handleAction(actionType: string, params: Record<string, any>) {
  switch (actionType) {
    case 'SCHEDULE_CALL':
      // Open calendar scheduling modal
      break;
    case 'SEND_EMAIL':
      // Open email composer
      break;
    case 'VIEW_PROFILE':
      // Navigate to stakeholder profile
      break;
    case 'VIEW_DETAILS':
      // Expand insight details panel
      break;
    case 'GENERATE_BRIEFING':
      // Trigger briefing generation
      break;
  }
}

// User dismisses an insight
await api.dismiss(dealId, insightId);

// User resolves an insight after taking action
await api.resolve(dealId, insightId, 'Called Arun, confirmed timeline for Q2.');
```

## Notes

- **AI Generation**: Insights are generated automatically during stakeholder analysis (background process). No manual trigger endpoint.
- **Expiration**: Some insights have `expires_at` set and will auto-transition to `EXPIRED` status. By default, expired insights are hidden.
- **Deal-Level Insights**: Some insights (like `MISSING_DECISION_MAKER`) are deal-level and have `stakeholder_id = null`.
- **Related Risks**: Some insights may link to Risk/Blocker items via `related_risk_blocker_id`.
